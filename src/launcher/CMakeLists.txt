cmake_minimum_required(VERSION 3.20)
project(DancingKittyDesktop VERSION 1.0.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUNDLE_GENERATED_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../build/bundle-generated" CACHE PATH "Recursos compilados")
add_executable(DancingKittyLauncher WIN32 bundle.cpp "${BUNDLE_GENERATED_DIR}/bundle.rc")
set_target_properties(DancingKittyLauncher PROPERTIES OUTPUT_NAME "Dancing Kitty")
target_include_directories(DancingKittyLauncher PRIVATE "${BUNDLE_GENERATED_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../app")
target_compile_definitions(DancingKittyLauncher PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(DancingKittyLauncher PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/utf-8;/W4;/O2;/EHsc>)
target_link_libraries(DancingKittyLauncher PRIVATE shell32 ole32)

cmake_minimum_required(VERSION 3.20)
project(DancingKitty VERSION 1.0.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CATPULSE_DEFAULT_CORE "${CMAKE_CURRENT_SOURCE_DIR}/../core")
set(KICKPULSE_CORE_DIR "${CATPULSE_DEFAULT_CORE}" CACHE PATH "Fuentes del motor KickPulse")
add_subdirectory("${KICKPULSE_CORE_DIR}" core)
add_executable(DancingKitty WIN32 cat_desktop.cpp cat_icon.rc)
set_target_properties(DancingKitty PROPERTIES OUTPUT_NAME "Dancing Kitty")
target_include_directories(DancingKitty PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(DancingKitty PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(DancingKitty PRIVATE /W4 /utf-8 /O2 /EHsc)
target_link_libraries(DancingKitty PRIVATE KickPulseCore windowscodecs gdi32 user32 shell32 psapi advapi32)
option(DANCINGKITTY_BUILD_TESTS "Animation clock tests" OFF)
if(DANCINGKITTY_BUILD_TESTS)
    enable_testing()
    add_executable(DancingKittyAnimationTests animation_tests.cpp)
    target_compile_options(DancingKittyAnimationTests PRIVATE /W4 /utf-8 /O2 /EHsc)
    add_test(NAME animation_clock COMMAND DancingKittyAnimationTests)
endif()

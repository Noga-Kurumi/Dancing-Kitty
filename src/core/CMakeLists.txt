cmake_minimum_required(VERSION 3.20)
project(KickPulseCore VERSION 0.12.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "KickPulse Core requiere Windows y MSVC")
endif()
set(ORT_ROOT "" CACHE PATH "Paquete Microsoft.ML.OnnxRuntime 1.23.2")

# Production library; validation tools are opt-in below.
add_library(KickPulseCore STATIC beat_engine.cpp)
target_include_directories(KickPulseCore PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" PRIVATE "${ORT_ROOT}/build/native/include")
target_compile_definitions(KickPulseCore PRIVATE UNICODE _UNICODE)
target_compile_options(KickPulseCore PRIVATE /W4 /utf-8 /O2 /fp:precise /EHsc)
target_link_libraries(KickPulseCore PUBLIC "${ORT_ROOT}/runtimes/win-x64/native/onnxruntime.lib" ole32 uuid avrt mmdevapi windowsapp normaliz)
option(KICKPULSE_BUILD_TESTS "Incluir pruebas del motor" OFF)
if(KICKPULSE_BUILD_TESTS)
    enable_testing()
    add_executable(KickPulseTests tests.cpp)
    target_compile_definitions(KickPulseTests PRIVATE UNICODE _UNICODE)
    target_compile_options(KickPulseTests PRIVATE /W4 /utf-8 /O2 /EHsc)
    add_test(NAME audio_stream COMMAND KickPulseTests)
    add_executable(KickPulseCatalogTests catalog_tests.cpp)
    target_compile_definitions(KickPulseCatalogTests PRIVATE UNICODE _UNICODE)
    target_compile_options(KickPulseCatalogTests PRIVATE /W4 /utf-8 /EHsc)
    target_link_libraries(KickPulseCatalogTests PRIVATE normaliz)
    add_test(NAME recording_matching COMMAND KickPulseCatalogTests)
    add_executable(KickPulseTempoTests tempo_tests.cpp)
    target_compile_options(KickPulseTempoTests PRIVATE /W4 /utf-8 /O2 /EHsc)
    add_test(NAME bpm_clock COMMAND KickPulseTempoTests)
    add_executable(KickPulseIntegrationProbe integration_probe.cpp)
    target_compile_options(KickPulseIntegrationProbe PRIVATE /W4 /utf-8 /O2 /EHsc)
    target_link_libraries(KickPulseIntegrationProbe PRIVATE KickPulseCore psapi)
    add_executable(KickPulseReplayProbe replay_probe.cpp)
    target_include_directories(KickPulseReplayProbe PRIVATE "${ORT_ROOT}/build/native/include")
    target_compile_options(KickPulseReplayProbe PRIVATE /W4 /utf-8 /O2 /EHsc)
    target_link_libraries(KickPulseReplayProbe PRIVATE "${ORT_ROOT}/runtimes/win-x64/native/onnxruntime.lib")
    add_executable(KickPulseCatalogProbe catalog_probe.cpp)
    target_compile_options(KickPulseCatalogProbe PRIVATE /W4 /utf-8 /O2 /EHsc)
    target_link_libraries(KickPulseCatalogProbe PRIVATE windowsapp normaliz)
    add_executable(KickPulseSourceTests source_tests.cpp)
    target_compile_options(KickPulseSourceTests PRIVATE /W4 /utf-8 /O2 /EHsc)
    target_link_libraries(KickPulseSourceTests PRIVATE windowsapp normaliz)
    add_test(NAME fallback_sources COMMAND KickPulseSourceTests)
    add_executable(KickPulseLocalTempoTests local_tempo_tests.cpp)
    target_compile_options(KickPulseLocalTempoTests PRIVATE /W4 /utf-8 /O2 /EHsc)
    add_test(NAME local_tempo COMMAND KickPulseLocalTempoTests)
endif()
